## 二、三个问题的精确定位

### 问题 1（编译直接报错）：`cmd_echo` 被定义了两次

编译器原话：

```
main.c:129:13: error: redefinition of 'cmd_echo'
main.c:120:13: note: previous definition of 'cmd_echo' with type 'void(int, char **)'
```

你的文件里有两个同名的函数：第 **120** 行和第 **129** 行。
C 语言里一个源文件不允许同名函数定义两次 → **删掉其中一个**即可。

建议：把 120-126 行那一小段（里面是 `printf("TODO: cmd_echo() 还没实现\n");` 和多余的 `};`）删掉，
保留下面那个带 `/* TODO 3 */` 注释的版本来实现。

### 问题 2（能编译但是逻辑错 + 警告）：`dispatch` 里的 `else` 放错了位置

你写的是：

```c
for (int i = 0; i < g_cmd_count; i++) {
    if (strcmp(argv[0], g_cmds[i].name) == 0) {
        g_cmds[i].handler(argc, argv);
        return 0;
    }
    else {                                    /* ← 问题在这里 */
        printf("未知命令: %s\n", argv[i]);
        printf("输入 help 看命令\n");
        return -1;
    }
}
```

**为什么错**：循环**第一次没比中**就立刻 `return -1` 了，后面的命令根本没机会被检查。
后果是——只有排在第一位的 `help` 能用，`led` / `echo` / `sys` 全都会被判成"未知命令"。

**正确的思路**：「没找到」是**循环跑完之后**才能下的结论，不是每次比较不中就能下的结论。
所以 `else` 要挪到循环外面：

```c
for (int i = 0; i < g_cmd_count; i++) {
    if (strcmp(g_cmds[i].name, argv[0]) == 0) {
        g_cmds[i].handler(argc, argv);
        return 0;                      /* 命中就返回，不再往下比 */
    }
}
/* 循环走完还没返回，说明一个都没匹配上 */
printf("未知命令: %s（输入 help 看命令）\n", argv[0]);
return -1;
```

另外还有两处小问题：

- `printf("未知命令: %s\n", argv[i])` 里的 `argv[i]` 应该改成 `argv[0]`。这里 `i` 是**命令表的索引**，不是参数索引；要报的是用户输入的命令名。
- 函数末尾少了一个 `return`，编译器给的是警告：
  `warning: control reaches end of non-void function [-]` —— 声明了返回 `int` 就必须保证所有路径都有返回值。

> 顺带说：`argc == 0` 那个分支你写的 `return 0`，骨架注释要求是"直接返回"，
> 改成 `return -1` 更合适（表示"这次没有命令被执行"）。`main` 里的 `argc == 0` 已经被 `continue` 拦掉了，所以影响不大，但语义上更准确。

## 三、完整参考实现

下面这份是**整份可编译、可运行**的版本，注释保留了骨架的风格，可以直接整段替换你的 `main.c`。
（建议先自己照着改，改不动再整体替换。）

```c
/*
 * W1D3 · 函数指针 + 命令表，实现命令行解析（参考实现）
 */

#include <stdio.h>
#include <string.h>

#define MAX_ARGS 8
#define MAX_LINE 128

typedef void (*cmd_handler_t)(int argc, char **argv);

typedef struct {
    const char   *name;
    cmd_handler_t handler;
    const char   *help;
} command_t;

static void cmd_help(int argc, char **argv);
static void cmd_echo(int argc, char **argv);
static void cmd_led(int argc, char **argv);
static void cmd_sys(int argc, char **argv);

/* 命令表：想加命令，只需要在这里加一行，分发逻辑一行都不用改 */
static const command_t g_cmds[] = {
    {"help", cmd_help, "help              列出所有命令"},
    {"echo", cmd_echo, "echo <文本...>    原样回显，用来验证切词"},
    {"led",  cmd_led,  "led on|off|toggle 控制 LED"},
    {"sys",  cmd_sys,  "sys               打印系统信息"},
};

static const int g_cmd_count = (int)(sizeof(g_cmds) / sizeof(g_cmds[0]));

static int g_led_on = 0;

/* ==================== TODO 1：切词 ==================== */
static int split_args(char *line, char **argv, int max_args)
{
    int   argc = 0;
    char *p    = line;

    while (*p != '\0' && argc < max_args) {
        while (*p == ' ' || *p == '\t') {           /* 跳过一段空白 */
            p++;
        }
        if (*p == '\0') {                           /* 后面全是空格，结束 */
            break;
        }

        argv[argc++] = p;                           /* 记住这一段从哪里开始 */

        while (*p != '\0' && *p != ' ' && *p != '\t') {
            p++;                                    /* 走到这一段结尾 */
        }
        if (*p != '\0') {
            *p = '\0';                              /* 就地写 '\0' 封口 */
            p++;                                    /* 越过空格，继续下一段 */
        }
    }
    return argc;
}

/* ==================== TODO 2：查表分发 ==================== */
static int dispatch(int argc, char **argv)
{
    if (argc == 0) {
        return -1;
    }

    for (int i = 0; i < g_cmd_count; i++) {
        if (strcmp(g_cmds[i].name, argv[0]) == 0) {   /* 命中 */
            g_cmds[i].handler(argc, argv);            /* 调用它的处理函数 */
            return 0;
        }
    }

    /* 循环跑完都没命中，才说明是未知命令 */
    printf("未知命令: %s（输入 help 看命令）\n", argv[0]);
    return -1;
}

/* ==================== 已给好的示例 ==================== */
static void cmd_help(int argc, char **argv)
{
    (void)argc;
    (void)argv;
    printf("可用命令：\n");
    for (int i = 0; i < g_cmd_count; i++) {
        printf("  %s\n", g_cmds[i].help);
    }
}

/* ==================== TODO 3：echo ==================== */
static void cmd_echo(int argc, char **argv)
{
    for (int i = 1; i < argc; i++) {
        printf("%s", argv[i]);
        if (i + 1 < argc) {
            printf(" ");        /* 最后一个参数后面不加空格 */
        }
    }
    printf("\n");
}

/* ==================== TODO 4：led ==================== */
static void cmd_led(int argc, char **argv)
{
    if (argc < 2) {
        printf("用法: led on|off|toggle\n");
        return;
    }

    if (strcmp(argv[1], "on") == 0) {
        g_led_on = 1;
    } else if (strcmp(argv[1], "off") == 0) {
        g_led_on = 0;
    } else if (strcmp(argv[1], "toggle") == 0) {
        g_led_on = !g_led_on;
    } else {
        printf("用法: led on|off|toggle\n");
        return;
    }

    printf("LED: %s\n", g_led_on ? "ON" : "OFF");
}

/* ==================== TODO 5：sys ==================== */
static void cmd_sys(int argc, char **argv)
{
    (void)argc;
    (void)argv;
    printf("[sys] 编译器   : GCC " __VERSION__ "\n");
    printf("[sys] 编译时间 : %s %s\n", __DATE__, __TIME__);
    printf("[sys] LED 状态 : %s\n", g_led_on ? "ON" : "OFF");
}

int main(void)
{
    char  line[MAX_LINE];
    char *argv[MAX_ARGS];

    printf("=== W1D3 命令行解析器 ===\n");
    printf("输入 help 看命令，输入 quit 退出。\n\n");

    while (1) {
        printf("w1d3> ");
        fflush(stdout);

        if (fgets(line, sizeof(line), stdin) == NULL) {
            break;
        }

        line[strcspn(line, "\r\n")] = '\0';

        int argc = split_args(line, argv, MAX_ARGS);
        if (argc == 0) {
            continue;
        }

        if (strcmp(argv[0], "quit") == 0 || strcmp(argv[0], "exit") == 0) {
            printf("bye\n");
            break;
        }

        dispatch(argc, argv);
    }

    return 0;
}
```

## 五、三个值得记进笔记的设计点

1. **表驱动**：命令名、处理函数、帮助文本都放在 `g_cmds` 表里；
   以后加命令只需要加一行，`dispatch` 和 `cmd_help` 一行都不用改。
   这就是为什么飞控固件里 MAVLink 消息分发、调试命令表都用这个结构。
2. **"没找到"是循环之后才能下的结论**：只要是在"遍历查找"的场景，
   判断"不存在"一定要放在循环外面——这个坑今天踩一次，以后就不会再踩。
3. **切词不复制数据**：`argv[i]` 指向的全是**原字符串内部**的位置，
   只是把空格换成了 `'\0'`。所以 `split_args` 的 `line` 必须是可修改的（数组），
   如果传进来的是字符串字面量（`const char *`）就会崩——这也是为什么 `main` 里用的是 `char line[MAX_LINE]`。
